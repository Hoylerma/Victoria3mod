# Erzeugt von werkzeuge/erzeuge_mod.py – nicht von Hand ändern.

# Wöchentliche Einnahmen minus Ausgaben des Staates
berater_wochenbilanz = {
	value = income
	subtract = expenses
}

# Anteil der Bevölkerung, der unter dem erwarteten Lebensstandard lebt (0 bis 1)
berater_anteil_unter_erwartung = {
	value = 0
	if = {
		limit = { total_population > 0 }
		value = population_below_expected_sol
		divide = total_population
	}
}

# Staats-Scope: ungefähre Zahl der Arbeitslosen
berater_arbeitslose = {
	value = state_population
	multiply = state_unemployment_rate
}

# Staats-Scope: Eignung als Bauort. Große Staaten mit vielen Arbeitslosen zuerst,
# Staaten mit überlasteter Infrastruktur nur halb so gut.
berater_staat_eignung = {
	value = state_unemployment_rate
	add = 0.1
	multiply = state_population
	if = {
		limit = { infrastructure_usage > infrastructure }
		multiply = 0.5
	}
}

# Staats-Scope: wie weit die Infrastruktur überlastet ist
berater_infra_defizit = {
	value = infrastructure_usage
	subtract = infrastructure
}

# Interessengruppen-Scope: Unmut (je höher, desto wütender)
berater_ig_unmut = {
	value = ig_approval
	multiply = -1
}
