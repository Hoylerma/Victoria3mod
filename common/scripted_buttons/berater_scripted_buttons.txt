berater_knopf_jahresbericht = {
	name = {
		first_valid = {
			triggered_desc = {
				desc = berater_knopf_jahresbericht_aus
				trigger = { has_variable = berater_kein_jahresbericht }
			}
			desc = berater_knopf_jahresbericht_an
		}
	}
	desc = berater_knopf_jahresbericht_desc
	visible = { always = yes }
	possible = { always = yes }
	effect = {
		if = {
			limit = { has_variable = berater_kein_jahresbericht }
			remove_variable = berater_kein_jahresbericht
		}
		else = {
			set_variable = berater_kein_jahresbericht
		}
	}
	ai_chance = {
		value = 0
	}
}
